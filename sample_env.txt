# ==============================================================================
# Core Environment Variables (used by multiple components)
# ==============================================================================

# Storage backend: SNOWFLAKE (push to Snowflake, default), BLOB (upload to Azure Blob
# Storage), or LOCAL (keep files locally, skip load). GLOFAS_THRESHOLD_SOURCE and
# GLOFAS_JRC_SOURCE below default to this same value, so switching this alone redirects
# everything the pipeline touches unless one of those is set explicitly to something
# different (a legitimate mixed-mode config, e.g. Blob for per-cycle output but Snowflake
# for those static reference files). When LOCAL, Snowflake credentials below are not
# required; when BLOB, Snowflake credentials are still not required, but the Blob
# credentials below are.
DATA_PIPELINE_DB=SNOWFLAKE

# Snowflake Connection (required when DATA_PIPELINE_DB=SNOWFLAKE, or when
# GLOFAS_THRESHOLD_SOURCE/GLOFAS_JRC_SOURCE=snowflake)
SNOWFLAKE_ACCOUNT=your_account_identifier
SNOWFLAKE_USER=your_username
SNOWFLAKE_PASSWORD=your_password
SNOWFLAKE_WAREHOUSE=your_warehouse
SNOWFLAKE_DATABASE=your_database
SNOWFLAKE_SCHEMA=your_schema

# Azure Blob Storage Connection (required when DATA_PIPELINE_DB=BLOB, or when
# GLOFAS_THRESHOLD_SOURCE/GLOFAS_JRC_SOURCE=blob)
ACCOUNT_URL=https://your_account.blob.core.windows.net/
SAS_TOKEN=your_container_scoped_sas_token
CONTAINER_NAME=your_container_name

# ==============================================================================
# Pipeline-specific (see github_actions/sample_env.txt for full reference)
# ==============================================================================

# Download tp + ro as Zarr to Snowflake stage (true/false, default: true)
PROCESS_MET=true

# Snowflake internal stage for met Zarr: REQUIRED when PROCESS_MET=true
SNOWFLAKE_STAGE_NAME=your_stage_name

# Enable gust envelope processing (true/false, default: true), independent of
# PROCESS_WIND_DATA
PROCESS_GUST=true

# Filter to named storms only (true/false, default: true)
NAMED_STORMS_ONLY=true

# ==============================================================================
# GloFAS riverine discharge (standalone pipeline)
# ==============================================================================

# Local dir for GloFAS forecast Zarr files
GLOFAS_DATA_DIR=glofas_data

# submit: fires the 2 CDS requests and exits (fast, no queue wait). process
# (default): resumes a prior submit's saved request IDs, or falls back to a
# fresh submit-and-block if nothing was saved.
GLOFAS_MODE=process

# Where the RP2 threshold file is read from: snowflake, blob, or local (fully offline
# dev, requires setup_glofas_thresholds.py --local-only to have been run first).
# Defaults to DATA_PIPELINE_DB above; set explicitly to use a different source for this
# static reference file than for the per-cycle output.
GLOFAS_THRESHOLD_SOURCE=snowflake

# Local dir containing rl_2.0.nc, used only when GLOFAS_THRESHOLD_SOURCE=local
GLOFAS_THRESHOLD_LOCAL_DIR=glofas_thresholds

# CDS/EWDS API credentials for cems-glofas-forecast (separate from ecmwf-opendata,
# no auth needed elsewhere in this pipeline). Format: https://cds.climate.copernicus.eu
# account page. Omit to fall back to a local ~/.cdsapirc file instead.
CDSAPI_URL=https://ewds.climate.copernicus.eu/api
CDSAPI_KEY=your_ewds_api_key

# ==============================================================================
# GloFAS x JRC extent-masking
# ==============================================================================

# Set to false to disable the extent-masking step entirely and only run raw
# discharge
GLOFAS_EXTENT_ENABLED=true

# Where the cached JRC RP10/20/50/100 + permanent-water GeoTIFFs are read from:
# snowflake, blob, or local (fully offline dev, requires setup_jrc_extents.py
# --local-only to have been run first). Defaults to DATA_PIPELINE_DB above, same as
# GLOFAS_THRESHOLD_SOURCE.
GLOFAS_JRC_SOURCE=snowflake

# Local dir containing jrc_rp10.tif etc., used only when GLOFAS_JRC_SOURCE=local
GLOFAS_JRC_LOCAL_DIR=glofas_jrc_extents

# Note: setup_jrc_extents.py (the one-time JRC cache-population script) fetches
# directly from JRC's own file server
